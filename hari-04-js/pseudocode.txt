1. Program Penentu Bilangan Ganjil/Genap dan Positif/Negatif/Nol (lab2_latihan1.js)
-----------------------------------------------------------------------------
ALGORITMA Penentu_Bilangan
DEKLARASI:
    nilai, statusGanjilGenap, statusTanda : Integer / String

DESKRIPSI:
    MULAI
    BACA nilai

    // Seleksi 1: Menentukan Ganjil atau Genap
    JIKA (nilai % 2 === 0) MAKA
        statusGanjilGenap = "genap"
    JIKA TIDAK
        statusGanjilGenap = "ganjil"
    AKHIR JIKA

    // Seleksi 2: Menentukan Positif, Negatif, atau Nol
    JIKA (nilai > 0) MAKA
        statusTanda = "positif"
    JIKA TIDAK, JIKA (nilai < 0) MAKA
        statusTanda = "negatif"
    JIKA TIDAK
        statusTanda = "nol"
    AKHIR JIKA

    TAMPILKAN statusGanjilGenap
    TAMPILKAN statusTanda
    SELESAI


2. Program Perhitungan Tarif Parkir (lab2_latihan2.js)
-----------------------------------------------------------------------------
ALGORITMA Hitung_Tarif_Parkir
DEKLARASI:
    durasiJam, sisaJam, totalHari, bayarHari, bayarTotal : Integer
    TARIF_AWAL = 3000          // 2 jam pertama
    TARIF_PER_JAM = 2000       // tiap jam berikutnya
    TARIF_MAKS_PER_HARI = 15000 // batas tarif per hari (24 jam)

DESKRIPSI:
    MULAI
    BACA durasiJam

    sisaJam = durasiJam % 24
    totalHari = (durasiJam - sisaJam) / 24

    // Hitung tarif sisa jam
    JIKA (sisaJam <= 2) MAKA
        bayarHari = TARIF_AWAL
    JIKA TIDAK
        bayarHari = TARIF_AWAL + (sisaJam - 2) * TARIF_PER_JAM
    AKHIR JIKA

    // Terapkan batas maksimal tarif per hari
    JIKA (bayarHari > TARIF_MAKS_PER_HARI) MAKA
        bayarHari = TARIF_MAKS_PER_HARI
    AKHIR JIKA

    bayarTotal = (totalHari * TARIF_MAKS_PER_HARI) + bayarHari
    TAMPILKAN bayarTotal
    SELESAI


3. Program Kalkulator Sederhana dengan Switch (lab2_latihan3.js & kalkulator.js)
-----------------------------------------------------------------------------
ALGORITMA Kalkulator_Sederhana
DEKLARASI:
    angka1, angka2 : Real
    operator : String
    hasil : Real / String

DESKRIPSI:
    MULAI
    BACA angka1
    BACA operator (+, -, *, /)
    BACA angka2

    VALIDASI: JIKA angka1 atau angka2 bukan angka MAKA
        TAMPILKAN "Input tidak valid"
        SELESAI
    AKHIR JIKA

    PILIH (operator)
        KASUS '+':
            hasil = angka1 + angka2
        KASUS '-':
            hasil = angka1 - angka2
        KASUS '*':
            hasil = angka1 * angka2
        KASUS '/':
            JIKA (angka2 === 0) MAKA
                hasil = "tidak terdefinisi (pembagian dengan nol)"
            JIKA TIDAK
                hasil = angka1 / angka2
            AKHIR JIKA
        DEFAULT:
            hasil = "Operator tidak valid"
    AKHIR PILIH

    TAMPILKAN hasil
    SELESAI


4. Program Penentu Kategori BMI (lab2_latihan4.js & lab4.html)
-----------------------------------------------------------------------------
ALGORITMA Penentu_Kategori_BMI
DEKLARASI:
    berat, tinggi, bmi : Real
    kategori : String

DESKRIPSI:
    MULAI
    BACA berat (kg)
    BACA tinggi (m)

    bmi = berat / (tinggi * tinggi)

    JIKA (bmi < 18.5) MAKA
        kategori = "kurang"
    JIKA TIDAK, JIKA (bmi <= 24.9) MAKA
        kategori = "normal"
    JIKA TIDAK, JIKA (bmi <= 29.9) MAKA
        kategori = "lebih"
    JIKA TIDAK
        kategori = "obesitas"
    AKHIR JIKA

    TAMPILKAN bmi
    TAMPILKAN kategori
    SELESAI
